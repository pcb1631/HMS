269int levenshtein_distance(const char* s, const char* t, int m, int n)
270{
271 int * v0 = malloc((n + 1) * sizeof(int));
272 int * v1 = malloc((n + 1) * sizeof(int));
273 int * swap;
274 int i;
275 int ii;
276 int result;
277
278 if(v0 == NULL OR v1 == NULL)
279 {
280 fprintf(stderr, "malloc fail levenshtein_distance()\n");
281 exit(1);
282 }
283
284 for(i = 0; i<n+1; i++)
285 v0[i] = i;
286
287 for(i=0; i<m; i++)
288 {
289 v1[0] = i+1;
290
291 for(ii=0; ii<n; ii++)
292 {
293 int del_cost = v0[ii+1]+1;
294 int insert_cost = v1[ii] + 1;
295 int sub_cost;
296
297 if (s[i]==t[ii])
298 sub_cost = v0[ii];
299 else
300 sub_cost = v0[ii]+1;
301
302 v1[ii + 1] = min(min(del_cost, sub_cost), insert_cost);
303 }
304
305 swap = v0;
306 v0 = v1;
307 v1 = swap;
308 }
309
310 result = v0[n];
311 free(v0); free(v1);
312 return result;
313}
